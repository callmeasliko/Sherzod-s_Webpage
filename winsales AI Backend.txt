// Necessary packages: express, cors, multer, dotenv, @google/generative-ai
const express = require('express');
const cors = require('cors');
const multer = require('multer');
const dotenv = require('dotenv');
const { GoogleGenerativeAI } = require("@google/generative-ai");
const fs = require('fs');

dotenv.config();

const app = express();
const port = process.env.PORT || 3000;

// Middleware configuration
app.use(cors()); // Allow requests from frontend
app.use(express.json());

// Setup storage for uploaded audio files
const upload = multer({ dest: 'uploads/' });

// Initialize Gemini AI
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY || "");

/**
 * Endpoint to analyze sales audio
 * POST /api/analyze-audio
 */
app.post('/api/analyze-audio', upload.single('audio'), async (req, res) => {
    try {
        const file = req.file;
        if (!file) {
            return res.status(400).send({ error: "Audio fayl yuklanmadi." });
        }

        // Convert uploaded file to base64 for Gemini
        const audioData = fs.readFileSync(file.path);
        const base64Audio = audioData.toString('base64');

        const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });

        const prompt = `Siz tajribali sotuv auditorisiz. Ushbu audio suhbatni eshiting. 
            Uni 'Top 50 sotuv xatolari' qoidalari asosida tahlil qiling. 
            1. Suhbatga 100 ballik tizimda baho bering.
            2. 3 ta eng katta xatoni ko'rsating.
            3. 2 ta yaxshilash maslahatini bering.
            Javobni O'zbek tilida bering.`;

        const result = await model.generateContent([
            prompt,
            {
                inlineData: {
                    data: base64Audio,
                    mimeType: file.mimetype
                }
            }
        ]);

        const responseText = result.response.text();

        // Clean up uploaded file
        fs.unlinkSync(file.path);

        res.send({ analysis: responseText });

    } catch (error) {
        console.error("Analysis Error:", error);
        res.status(500).send({ error: "Tahlil jarayonida xatolik yuz berdi." });
    }
});

/**
 * Simple simulator endpoint
 * POST /api/chat
 */
app.post('/api/chat', async (req, res) => {
    try {
        const { message, persona } = req.body;
        const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });

        const systemPrompt = `Siz ${persona} xarakteridagi mijozsiz. 
            Menejer sizga nimanidir sotmoqchi. Qisqa va e'tirozli javob bering. O'zbek tilida.`;

        const result = await model.generateContent([systemPrompt, message]);
        res.send({ reply: result.response.text() });

    } catch (error) {
        res.status(500).send({ error: "Chatda xatolik." });
    }
});

app.listen(port, () => {
    console.log(`Backend server running at http://localhost:${port}`);
});
